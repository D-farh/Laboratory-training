/*==============================================================================
 * 项目：两路红外循迹小车
 * 平台：8051（STC89C52 / AT89C52 等）+ L298N 双 H 桥驱动
 *
 * 引脚分配
 *   P2.1  SEN_OUT1   左循迹传感器输出
 *   P2.0  SEN_OUT2   右循迹传感器输出
 *   P1.0  L298_IN1   左电机方向位 1
 *   P1.1  L298_IN2   左电机方向位 2
 *   P1.2  L298_IN3   右电机方向位 1
 *   P1.3  L298_IN4   右电机方向位 2
 *   P1.4  L298_ENA   左电机使能（PWM 调速）
 *   P1.5  L298_ENB   右电机使能（PWM 调速）
 *
 * 传感器电平约定
 *   0 = 白底（未压线）    1 = 黑线（压线）
 *   若你的模块输出相反，把 Track_Scan() 里的 0/1 对调即可。
 *============================================================================*/
#include <reg51.h>

/*--------------------------- 引脚定义 ---------------------------*/
sbit SEN_OUT1 = P2^1;      /* 左循迹传感器 */
sbit SEN_OUT2 = P2^0;      /* 右循迹传感器 */

sbit L298_ENA = P1^4;      /* 左电机使能端，输入 PWM 波调速 */
sbit L298_IN1 = P1^0;      /* 左电机方向控制 */
sbit L298_IN2 = P1^1;
sbit L298_IN3 = P1^2;      /* 右电机方向控制 */
sbit L298_IN4 = P1^3;
sbit L298_ENB = P1^5;      /* 右电机使能端，输入 PWM 波调速 */

/*--------------------------- PWM 参数 ---------------------------*/
#define T 100              /* 一个 PWM 周期分成 100 份，占空比 = PWMx / 100 */

unsigned char PWML, PWMR;  /* 左、右电机占空比，取值范围 0~100 */

/*------------------------------------------------------------------
 * 函数：Timer0_Init
 * 功能：定时器0 初始化 —— 方式1（16 位定时器），并使能中断
 *----------------------------------------------------------------*/
void Timer0_Init()
{
    TMOD |= 0x01;          /* 只把 T0 设为方式1(M1M0=01)，不影响 T1 的配置 */
    TH0 = 0xFF;            /* 装初值：决定每"份"PWM 的时间（此值偏小，见文末说明） */
    TL0 = 0xFF;
    ET0 = 1;               /* 允许定时器0 中断 */
    TR0 = 1;               /* 启动定时器0 开始计数 */
    EA  = 1;               /* 打开总中断 */
}

/*------------------------------------------------------------------
 * 函数：Timer0_ISR
 * 功能：定时器0 中断服务程序，用软件方式产生两路 PWM
 *
 * 原理：每进一次中断，份数 cnt 加 1，cnt 在 0~99 之间循环，
 *       相当于把每个 PWM 周期切成 100 份。
 *       当 cnt < PWML 时左使能脚输出高电平，于是高电平占的份数
 *       就是 PWML 份，占空比 = PWML / 100，实现"数字调速"。
 *----------------------------------------------------------------*/
void Timer0_ISR() interrupt 1
{
    static unsigned char cnt = 0;   /* 当前处在 PWM 周期的第几份，0~T-1 */

    TH0 = 0xFF;            /* 重新装载初值，否则下次中断时间会错 */
    TL0 = 0xFF;

    cnt++;                 /* 进入下一份 */
    if(cnt >= T) cnt = 0;  /* 满 100 份，一个 PWM 周期结束，从头开始 */

    /* 比较得到两路占空比不同的方波，分别驱动左右电机 */
    L298_ENA = (cnt < PWML) ? 1 : 0;
    L298_ENB = (cnt < PWMR) ? 1 : 0;
}

/*------------------------------------------------------------------
 * 函数：Car_Forward
 * 功能：直行 —— 左右电机同向、同速
 * 说明：IN1=1,IN2=0 左电机正转；IN3=1,IN4=0 右电机正转
 *       占空比 59% 表示约 59% 的电源电压加在电机上
 *----------------------------------------------------------------*/
void Car_Forward(void)
{
    L298_IN1 = 1;
    L298_IN2 = 0;
    L298_IN3 = 1;
    L298_IN4 = 0;
    PWML = 59;             /* 左轮 59% 速度 */
    PWMR = 59;             /* 右轮 59% 速度 */
}

/*------------------------------------------------------------------
 * 函数：Car_SlightRight
 * 功能：向右修正 —— 左轮快、右轮慢，靠左右轮速差转弯
 * 说明：左 95%、右 3%，右轮几乎停住，实际是"急右转"而不是微调
 *----------------------------------------------------------------*/
void Car_SlightRight(void)
{
    L298_IN1 = 1;
    L298_IN2 = 0;
    L298_IN3 = 1;
    L298_IN4 = 0;
    PWML = 95;             /* 左轮全速前进 */
    PWMR = 3;              /* 右轮几乎不动 */
}

/*------------------------------------------------------------------
 * 函数：Car_SlightLeft
 * 功能：向左修正 —— 左轮慢、右轮快
 *----------------------------------------------------------------*/
void Car_SlightLeft(void)
{
    L298_IN1 = 1;
    L298_IN2 = 0;
    L298_IN3 = 1;
    L298_IN4 = 0;
    PWML = 3;              /* 左轮几乎不动 */
    PWMR = 95;             /* 右轮全速前进 */
}

/*------------------------------------------------------------------
 * 函数：Car_Stop
 * 功能：停车 —— 两路 PWM 都置 0，使能脚恒为低
 * 说明：EN=0 时 L298 输出为高阻，电机是"自由滑行"而不是刹车；
 *       方向脚 IN1~IN4 保持原来的电平不变。
 *----------------------------------------------------------------*/
void Car_Stop(void)
{
    PWML = 0;
    PWMR = 0;
}

/*------------------------------------------------------------------
 * 函数：Track_Scan
 * 功能：读取两路传感器，判断小车与黑线的相对位置并执行动作
 *
 *   SEN_OUT1(左)  SEN_OUT2(右)   情况                  动作
 *   ---------------------------------------------------------------
 *        0            0          线在中间 / 丢线        直行
 *        0            1          右传感器压线，车偏左   右转（左快右慢）
 *        1            0          左传感器压线，车偏右   左转（左慢右快）
 *        1            1          两传感器都压线         停车
 *----------------------------------------------------------------*/
void Track_Scan(void)
{
    /* 两边都没压线：认为线在车身正中间，直行 */
    if(SEN_OUT1 == 0 && SEN_OUT2 == 0)
    {
        Car_Forward();
    }
    /* 只有右边压线：说明车身偏左了，向右修正 */
    else if(SEN_OUT1 == 0 && SEN_OUT2 == 1)
    {
        Car_SlightRight();
    }
    /* 只有左边压线：说明车身偏右了，向左修正 */
    else if(SEN_OUT1 == 1 && SEN_OUT2 == 0)
    {
        Car_SlightLeft();
    }
    /* 两边都压线：遇到十字/横线，停车（也可改为继续直行通过） */
    else if(SEN_OUT1 == 1 && SEN_OUT2 == 1)
    {
        Car_Stop();
    }
}

/*------------------------------------------------------------------
 * 函数：main
 * 功能：初始化定时器后进入死循环，不断采样、判断、控制
 * 说明：PWM 由定时器中断在后台产生，主循环只负责"决策"
 *----------------------------------------------------------------*/
void main(void)
{
    Timer0_Init();         /* 启动 PWM 节拍 */

    while(1)
    {
        Track_Scan();      /* 反复循迹，周期越短响应越灵敏 */
    }
}
